/// همهٔ جفت‌های متداخل را در O(n log n) برمی‌گرداند.
/// ورودی باید [start, end) باشد — انتهای باز، تا کارهای پشت‌سرهم تداخل حساب نشوند.
List<(T, T)> overlappingPairs<T>(
  List<T> items, {
  required DateTime Function(T) start,
  required DateTime Function(T) end,
}) {
  final sorted = [...items]..sort((a, b) => start(a).compareTo(start(b)));
  final out = <(T, T)>[];
  final active = <T>[];               // مرتب بر اساس end
  for (final it in sorted) {
    active.removeWhere((a) => !end(a).isAfter(start(it)));
    for (final a in active) out.add((a, it));
    active.add(it);
  }
  return out;
}

T-9.3 · امتیازدهندهٔ یادگیرنده (م‌۶)
رگرسیون لجستیک تک‌لایه، کاملاً روی دستگاه (قانون ۲۸ · قانون ۱۴ — بدون پکیج):
class LearnedScorer {
  static const _eta = 0.05;           // نرخ یادگیری — RitmoConstants
  static const _wMin = -2.0, _wMax = 2.0;

  List<double> w;                     // طول = EvidenceCode.values.length

  double predict(List<double> x) =>
      1 / (1 + exp(-_dot(w, x)));     // 0..1

  /// y = 1 اگر کاربر انجامش داد، 0 اگر رد/نادیده گرفت
  void learn(List<double> x, int y) {
    final p = predict(x);
    for (var i = 0; i < w.length; i++) {
      w[i] = (w[i] + _eta * (y - p) * x[i]).clamp(_wMin, _wMax);
    }
  }
}

فایل lib/core/domain/engines/engine_flight_recorder.dart:
class EngineFlightRecorder {
  static const _capacity = 200;           // حلقوی — حافظه ثابت
  final Queue<EngineRun> _ring = Queue();

  void record(EngineRun r) {
    _ring.addLast(r);
    if (_ring.length > _capacity) _ring.removeFirst();
  }

  List<EngineRun> get runs => List.unmodifiable(_ring);
}

class EngineRun {
  final Type engine;
  final String fingerprint;
  final bool cacheHit;
  final int durationMs;
  final int queryCount;
  final DateTime at;
  final String? errorType;   // فقط نوع خطا، نه دادهٔ کاربر
}
